-- LOGIKA KARAKTER: God Mode (Kebal) & Speed Hack (Kecepatan)
----------------------------------------------------------------
local GOD_HP = 1e9 -- darah "tak terbatas" (pakai angka besar, bukan math.huge, supaya bar darah tidak error NaN)
SPEED_MIN, SPEED_MAX, SPEED_DEFAULT = 4, 200, 50 -- rentang kecepatan slider (bawaan Roblox adalah 16)

local godOn, speedOn = false, false
local speedValue = SPEED_DEFAULT
local baseMaxHealth, baseWalkSpeed = 100, 16 -- nilai asli game, dipakai untuk mengembalikan saat dimatikan
local godConns, speedConn = {}, nil

JUMP_MIN, JUMP_MAX, JUMP_DEFAULT = 10, 300, 100 -- rentang kekuatan lompat slider (bawaan Roblox JumpPower = 50)
local jumpOn, fullOn = false, false
local jumpValue = JUMP_DEFAULT
local baseUseJump, baseJumpPower, baseJumpHeight = true, 50, 7.2 -- nilai asli game
local jumpConns, fullConn = {}, nil

local invisOn = false -- Tak Terlihat
local invisConns, invisOrig = {}, {} -- koneksi event / Transparency asli tiap bagian tubuh
local baseDisplayDist = Enum.HumanoidDisplayDistanceType.Viewer -- tipe tampilan nama + bar darah asli

local noclipOn = false -- Tembus Dinding
local noclipConn = nil -- koneksi Stepped (berjalan tiap frame fisika)
local noclipOrig = setmetatable({}, { __mode = "k" }) -- bagian tubuh yang CanCollide-nya dimatikan (untuk dikembalikan saat dimatikan)

function getHumanoid()
	local char = player.Character
	return char and char:FindFirstChildOfClass("Humanoid")
end

local function clearGodConns()
	for _, c in ipairs(godConns) do c:Disconnect() end
	table.clear(godConns)
end

local function startGod(hum)
	clearGodConns()
	if hum.MaxHealth < GOD_HP then baseMaxHealth = hum.MaxHealth end
	hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false) -- karakter tidak bisa mati
	if not hum.Parent:FindFirstChild("VuDongGodFF") then
		local ff = Instance.new("ForceField") -- blokir damage, sembunyikan efek
		ff.Name, ff.Visible = "VuDongGodFF", false
		ff.Parent = hum.Parent
	end
	hum.MaxHealth = GOD_HP
	hum.Health = GOD_HP
	table.insert(godConns, hum.HealthChanged:Connect(function(h)
		if godOn and h < hum.MaxHealth then hum.Health = hum.MaxHealth end -- darah berkurang langsung dipulihkan
	end))
	table.insert(godConns, hum:GetPropertyChangedSignal("MaxHealth"):Connect(function()
		if godOn and hum.MaxHealth ~= GOD_HP then
			hum.MaxHealth = GOD_HP
			hum.Health = GOD_HP
		end
	end))
end

local function stopGod(hum)
	clearGodConns()
	if hum and hum.Parent then
		local ff = hum.Parent:FindFirstChild("VuDongGodFF")
		if ff then ff:Destroy() end
		hum:SetStateEnabled(Enum.HumanoidStateType.Dead, true)
		hum.MaxHealth = baseMaxHealth
		hum.Health = baseMaxHealth
	end
end

local function startSpeed(hum)
	if speedConn then speedConn:Disconnect() end
	baseWalkSpeed = hum.WalkSpeed
	hum.WalkSpeed = speedValue
	speedConn = hum:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
		if speedOn and hum.WalkSpeed ~= speedValue then hum.WalkSpeed = speedValue end -- kalau game mengubah kecepatan, kembalikan lagi
	end)
end

local function stopSpeed(hum)
	if speedConn then
		speedConn:Disconnect()
		speedConn = nil
	end
	if hum and hum.Parent then hum.WalkSpeed = baseWalkSpeed end
end

-- slider memanggil fungsi ini; jika Speed Hack sedang aktif, langsung diterapkan
function setSpeedValue(v)
	speedValue = math.clamp(math.floor(v + 0.5), SPEED_MIN, SPEED_MAX)
	if speedOn then
		local hum = getHumanoid()
		if hum then hum.WalkSpeed = speedValue end
	end
end

local function clearJumpConns()
	for _, c in ipairs(jumpConns) do c:Disconnect() end
	table.clear(jumpConns)
end

local function startJump(hum)
	clearJumpConns()
	baseUseJump, baseJumpPower, baseJumpHeight = hum.UseJumpPower, hum.JumpPower, hum.JumpHeight
	hum.UseJumpPower = true -- jika game memakai JumpHeight, ganti ke JumpPower agar slider berfungsi
	hum.JumpPower = jumpValue
	local function keep() -- kalau game mengubahnya, kembalikan sesuai slider
		if not jumpOn then return end
		if not hum.UseJumpPower then hum.UseJumpPower = true end
		if hum.JumpPower ~= jumpValue then hum.JumpPower = jumpValue end
	end
	table.insert(jumpConns, hum:GetPropertyChangedSignal("JumpPower"):Connect(keep))
	table.insert(jumpConns, hum:GetPropertyChangedSignal("UseJumpPower"):Connect(keep))
end

local function stopJump(hum)
	clearJumpConns()
	if hum and hum.Parent then
		hum.UseJumpPower = baseUseJump
		hum.JumpPower = baseJumpPower
		hum.JumpHeight = baseJumpHeight
	end
end

-- slider memanggil fungsi ini; jika Jump Power sedang aktif, langsung diterapkan
function setJumpValue(v)
	jumpValue = math.clamp(math.floor(v + 0.5), JUMP_MIN, JUMP_MAX)
	if jumpOn then
		local hum = getHumanoid()
		if hum then hum.JumpPower = jumpValue end
	end
end

	fullConn = hum.HealthChanged:Connect(function(h)
		if fullOn and h > 0 and h < hum.MaxHealth then hum.Health = hum.MaxHealth end -- darah berkurang langsung dipenuhi lagi
	end)
end

local function stopFull()

	end
end

-- Tak Terlihat: set Transparency = 1 untuk semua bagian tubuh, aksesori, wajah (Decal) dan sembunyikan nama / bar darah di atas kepala.
-- Nilai asli disimpan untuk dikembalikan saat dimatikan. Benda baru yang muncul (aksesori, tool yang dipegang) juga otomatis disembunyikan.
-- CATATAN: LocalScript hanya mengubah sisi pemain ini; agar pemain lain juga tidak melihat, perlu Script + RemoteEvent di sisi server.
local function clearInvisConns()
	for _, c in ipairs(invisConns) do c:Disconnect() end
	table.clear(invisConns)
end

local function hideInvisPart(inst)
	if invisOrig[inst] ~= nil then return end -- sudah diproses
	local isPart = inst:IsA("BasePart") and inst.Name ~= "HumanoidRootPart"
	if not (isPart or inst:IsA("Decal")) then return end -- Decal mencakup Texture / wajah karakter
	invisOrig[inst] = inst.Transparency
	inst.Transparency = 1
	table.insert(invisConns, inst:GetPropertyChangedSignal("Transparency"):Connect(function()
		if invisOn and inst.Transparency ~= 1 then inst.Transparency = 1 end -- kalau game mengubahnya, sembunyikan lagi
	end))
end

local function startInvis(char, hum)
	clearInvisConns()
	table.clear(invisOrig)
	for _, d in ipairs(char:GetDescendants()) do
		hideInvisPart(d)
	end
	table.insert(invisConns, char.DescendantAdded:Connect(function(d)
		if invisOn then hideInvisPart(d) end
	end))
	if hum then
		baseDisplayDist = hum.DisplayDistanceType
		hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None -- sembunyikan nama + bar darah
	end
end

local function stopInvis(hum)
	clearInvisConns() -- putuskan koneksi dulu, baru kembalikan nilai asli
	for inst, t in pairs(invisOrig) do
		if inst.Parent then inst.Transparency = t end
	end
	table.clear(invisOrig)
	if hum and hum.Parent then hum.DisplayDistanceType = baseDisplayDist end
end

-- Tembus Dinding (Noclip): matikan CanCollide semua bagian tubuh tiap frame fisika (Humanoid sering menyalakannya
-- kembali sendiri, jadi harus diset terus-menerus). Hanya bagian yang dimatikan script yang diingat, agar status dikembalikan dengan benar saat dimatikan.
-- Loop membaca player.Character setiap kali, jadi tetap berfungsi setelah respawn tanpa perlu dipasang ulang.
local function startNoclip()
	if noclipConn then noclipConn:Disconnect() end
	noclipConn = RunService.Stepped:Connect(function()
		local char = player.Character
		if not (noclipOn and char) then return end
		for _, d in ipairs(char:GetDescendants()) do
			if d:IsA("BasePart") and d.CanCollide then
				noclipOrig[d] = true
				d.CanCollide = false
			end
		end
	end)
end

local function stopNoclip()
	if noclipConn then
		noclipConn:Disconnect()
		noclipConn = nil
	end
	for part in pairs(noclipOrig) do
		if part.Parent then part.CanCollide = true end
	end
	table.clear(noclipOrig)
end

-- LOGIKA ESP: melihat pemain lain menembus dinding
-- (Highlight garis tepi + warna, serta Nama / Darah / Jarak di atas kepala)

ESP_COLOR_NAMES = { "Merah", "Hijau", "Kuning", "Biru", "Putih" }
ESP_COLORS = {
	["Merah"] = Color3.fromRGB(255, 60, 60),
	["Hijau"] = Color3.fromRGB(60, 230, 90),
	["Kuning"] = Color3.fromRGB(255, 220, 50),
	["Biru"] = Color3.fromRGB(50, 170, 255),
	["Putih"] = Color3.fromRGB(255, 255, 255),
}
espColor = ESP_COLORS["Merah"]

local espHL, espName, espHP, espDist = false, false, false, false -- 4 sakelar ESP
local espObjs = {} -- espObjs[plr] = { hl = Highlight, gui = BillboardGui, txt = TextLabel }
local espPlrConns = {} -- espPlrConns[plr] = koneksi CharacterAdded milik pemain itu
local espGlobalConns = {} -- PlayerAdded / PlayerRemoving
local espBeat, espLast = nil, 0

local function espClear(plr)
	local o = espObjs[plr]
	if o then
		if o.hl then o.hl:Destroy() end
		if o.gui then o.gui:Destroy() end
		espObjs[plr] = nil
	end
end

local function espBuild(plr)
	espClear(plr)
	local char = plr.Character
	if not char then return end
	local head = char:FindFirstChild("Head") or char:WaitForChild("Head", 5)
	if not head or plr.Character ~= char then return end

	local hl = Instance.new("Highlight")
	hl.Name = "VuDongESP"
	hl.Adornee = char
	hl.FillColor, hl.OutlineColor = espColor, Color3.new(1, 1, 1)
	hl.FillTransparency, hl.OutlineTransparency = 0.6, 0
	hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop -- tampil menembus dinding
	hl.Enabled = espHL
	hl.Parent = char

	local gui = Instance.new("BillboardGui")
	gui.Name = "VuDongESPInfo"
	gui.Adornee = head
	gui.AlwaysOnTop = true
	gui.Size = UDim2.fromOffset(220, 70)
	gui.StudsOffset = Vector3.new(0, 3, 0)
	gui.Enabled = espName or espHP or espDist
	gui.Parent = head

	local txt = Instance.new("TextLabel")
	txt.BackgroundTransparency = 1
	txt.Size = UDim2.fromScale(1, 1)
	txt.Font = Enum.Font.SourceSansBold
	txt.TextSize = 16
	txt.RichText = true
	txt.TextColor3 = espColor
	txt.TextStrokeTransparency = 0.3
	txt.Text = ""
	txt.Parent = gui

	espObjs[plr] = { hl = hl, gui = gui, txt = txt }
end

-- perbarui hidup/mati tiap bagian + isi teks (dipanggil ~10 kali/detik)
local function espUpdate()
	local myHum = getHumanoid()
	local myRoot = myHum and myHum.RootPart
	local showGui = espName or espHP or espDist
	for plr, o in pairs(espObjs) do
		local char = plr.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if hum and o.gui.Parent then
			o.hl.Enabled = espHL
			o.hl.FillColor = espColor
			o.gui.Enabled = showGui
			if showGui then
				o.txt.TextColor3 = espColor
				local lines = {}
				if espName then
					table.insert(lines, plr.DisplayName)
				end
				if espHP then
					local ratio = hum.MaxHealth > 0 and math.clamp(hum.Health / hum.MaxHealth, 0, 1) or 0
					local col = Color3.fromRGB(255, 60, 60):Lerp(Color3.fromRGB(60, 230, 90), ratio)
					table.insert(lines, '<font color="#' .. col:ToHex() .. '">❤ ' .. math.floor(hum.Health) .. "</font>")
				end
				if espDist and myRoot and hum.RootPart then
					table.insert(lines, math.floor((hum.RootPart.Position - myRoot.Position).Magnitude) .. " m")
				end
				o.txt.Text = table.concat(lines, "\n")
			end
		end
	end
end

local function espHook(plr)
	if plr == player or espPlrConns[plr] then return end
	espPlrConns[plr] = plr.CharacterAdded:Connect(function()
		task.spawn(espBuild, plr)
	end)
	if plr.Character then task.spawn(espBuild, plr) end
end

local function espStopAll()
	for _, c in ipairs(espGlobalConns) do c:Disconnect() end
	table.clear(espGlobalConns)
	for _, c in pairs(espPlrConns) do c:Disconnect() end
	table.clear(espPlrConns)
	for plr in pairs(espObjs) do espClear(plr) end
	if espBeat then
		espBeat:Disconnect()
		espBeat = nil
	end
end

-- dipanggil setiap sakelar ESP diubah: selama masih ada yang aktif tetap berjalan, kalau semua mati dibersihkan
local function espRefresh()
	if espHL or espName or espHP or espDist then
		if not espBeat then
			for _, p in ipairs(Players:GetPlayers()) do espHook(p) end
			table.insert(espGlobalConns, Players.PlayerAdded:Connect(espHook))
			table.insert(espGlobalConns, Players.PlayerRemoving:Connect(function(plr)
				local c = espPlrConns[plr]
				if c then
					c:Disconnect()
					espPlrConns[plr] = nil
				end
				espClear(plr)
			end))
			espBeat = RunService.Heartbeat:Connect(function()
				local now = os.clock()
				if now - espLast < 0.1 then return end
				espLast = now
				espUpdate()
			end)
		end
		espUpdate()
	else
		espStopAll()
	end
end

----------------------------------------------------------------
-- LOGIKA ESP PETI: menandai SEMUA peti di map (garis tepi kuning + nama + jarak, menembus dinding)
-- Cara mengenali: objek (Model / Part) yang namanya mengandung salah satu dari CHEST_KEYWORDS.
-- >>> Kalau peti di game kamu namanya lain, ubah CHEST_KEYWORDS sesuai nama itu. <<<
-- CATATAN: jika game mengaktifkan StreamingEnabled, peti yang jauh (belum dimuat) belum tampil sampai kamu mendekat.
----------------------------------------------------------------
local CHEST_KEYWORDS = { "chest", "peti", "rương", "ruong" } -- huruf kecil; dicari dengan cara "mengandung" (kata Vietnam dipertahankan agar cocok dengan nama di game; "peti" ditambahkan)
local CHEST_IGNORE = { "plate", "armor", "armour" } -- abaikan nama yang mengandung kata ini (mis. ChestPlate = baju zirah)
local CHEST_COLOR = Color3.fromRGB(255, 200, 40)

local chestOn = false
local chestObjs = {} -- chestObjs[inst] = { hl = Highlight, gui = BillboardGui, txt = TextLabel, part = BasePart }
local chestConns = {}
local chestBeat, chestLast = nil, 0

local function chestMatch(name)
	local l = string.lower(name)
	for _, bad in ipairs(CHEST_IGNORE) do
		if string.find(l, bad, 1, true) then return false end
	end
	for _, kw in ipairs(CHEST_KEYWORDS) do
		if string.find(l, kw, 1, true) then return true end
	end
	return false
end

local function chestClear(inst)
	local o = chestObjs[inst]
	if o then
		if o.hl then o.hl:Destroy() end
		if o.gui then o.gui:Destroy() end
		chestObjs[inst] = nil
	end
end

local function chestAdd(inst)
	if chestObjs[inst] then return end
	if not (inst:IsA("Model") or inst:IsA("BasePart")) then return end
	if not chestMatch(inst.Name) then return end
	-- kalau berada di dalam peti yang sudah ditandai (mis. tutup peti), lewati agar tidak ganda
	local a = inst.Parent
	while a and a ~= workspace do
		if chestObjs[a] then return end
		a = a.Parent
	end
	-- jangan tandai benda yang ada pada pemain
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr.Character and inst:IsDescendantOf(plr.Character) then return end
	end
	local part = inst:IsA("BasePart") and inst or inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart", true)
	if not part then return end

	local hl = Instance.new("Highlight")
	hl.Name = "VDMark"
	hl.Adornee = inst
	hl.FillColor, hl.OutlineColor = CHEST_COLOR, Color3.new(1, 1, 1)
	hl.FillTransparency, hl.OutlineTransparency = 0.5, 0
	hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop -- tampil menembus dinding
	hl.Parent = inst

	local gui = Instance.new("BillboardGui")
	gui.Name = "VDMarkInfo"
	gui.Adornee = part
	gui.AlwaysOnTop = true
	gui.Size = UDim2.fromOffset(200, 50)
	gui.StudsOffset = Vector3.new(0, 3, 0)
	gui.Parent = part

	local txt = Instance.new("TextLabel")
	txt.BackgroundTransparency = 1
	txt.Size = UDim2.fromScale(1, 1)
	txt.Font = Enum.Font.SourceSansBold
	txt.TextSize = 16
	txt.TextColor3 = CHEST_COLOR
	txt.TextStrokeTransparency = 0.3
	txt.Text = "📦 " .. inst.Name
	txt.Parent = gui

	chestObjs[inst] = { hl = hl, gui = gui, txt = txt, part = part }
end

-- perbarui jarak (~4 kali/detik) dan bersihkan peti yang sudah hilang
local function chestUpdate()
	local myHum = getHumanoid()
	local myRoot = myHum and myHum.RootPart
	for inst, o in pairs(chestObjs) do
		if inst.Parent and o.part.Parent then
			local t = "📦 " .. inst.Name
			if myRoot then
				t = t .. "\n" .. math.floor((o.part.Position - myRoot.Position).Magnitude) .. " m"
			end
			o.txt.Text = t
		else
			chestClear(inst)
		end
	end
end

local function chestStop()
	for _, c in ipairs(chestConns) do c:Disconnect() end
	table.clear(chestConns)
	if chestBeat then
		chestBeat:Disconnect()
		chestBeat = nil
	end
	for inst in pairs(chestObjs) do chestClear(inst) end
end

local function chestRefresh()
	if chestOn then
		if chestBeat then return end
		table.insert(chestConns, workspace.DescendantAdded:Connect(function(d)
			if chestOn then chestAdd(d) end
		end))
		table.insert(chestConns, workspace.DescendantRemoving:Connect(function(d)
			if chestObjs[d] then chestClear(d) end
		end))
		chestBeat = RunService.Heartbeat:Connect(function()
			local now = os.clock()
			if now - chestLast < 0.25 then return end
			chestLast = now
			chestUpdate()
		end)
		-- pindai seluruh map (dipecah kecil agar tidak lag)
		task.spawn(function()
			local n = 0
			for _, d in ipairs(workspace:GetDescendants()) do
				if not chestOn then return end
				chestAdd(d)
				n += 1
				if n % 1500 == 0 then task.wait() end
			end
		end)
	else
		chestStop()
	end
end

-- LOGIKA: Terbang (Fly) - diadaptasi dari FlyGuiV3 (XNEO)
-- Cara terbang versi asli dipertahankan: BodyGyro + BodyVelocity menahan karakter melayang dan menghadap sesuai kamera,
-- mematikan state Humanoid + animasi diam, bergerak mengikuti arah MoveDirection, UP / DOWN menggeser
-- HumanoidRootPart naik / turun, slider = angka "speeds" (1 = bawaan versi asli).
-- Beda dari versi asli: tanpa variabel global, hanya SATU loop Heartbeat (versi asli menambah loop baru tiap klik +/-),
-- memakai dt sehingga kecepatan tidak berubah saat Boost FPS 120 aktif, dipasang di HumanoidRootPart (R6 / R15 sama-sama bisa),
-- mengembalikan state Humanoid semula saat dimatikan, dan otomatis terbang lagi setelah respawn.
----------------------------------------------------------------
local fly = { on = false, speed = 1, up = false, down = false, conn = nil, bg = nil, bv = nil, saved = {} }
local FLY_MIN, FLY_MAX, FLY_DEFAULT = 1, 20, 1 -- rentang kecepatan slider (1 = kecepatan asli FlyGuiV3)
local FLY_STATES = {
	Enum.HumanoidStateType.Climbing, Enum.HumanoidStateType.FallingDown, Enum.HumanoidStateType.Flying,
	Enum.HumanoidStateType.Freefall, Enum.HumanoidStateType.GettingUp, Enum.HumanoidStateType.Jumping,
	Enum.HumanoidStateType.Landed, Enum.HumanoidStateType.Physics, Enum.HumanoidStateType.PlatformStanding,
	Enum.HumanoidStateType.Ragdoll, Enum.HumanoidStateType.Running, Enum.HumanoidStateType.RunningNoPhysics,
	Enum.HumanoidStateType.Seated, Enum.HumanoidStateType.StrafingNoPhysics, Enum.HumanoidStateType.Swimming,
}
fly.speed = FLY_DEFAULT

local function flyClean() -- lepas BodyGyro / BodyVelocity
	if fly.bg then fly.bg:Destroy() fly.bg = nil end
	if fly.bv then fly.bv:Destroy() fly.bv = nil end
end

local function flyTracks(hum, speed) -- diamkan / kembalikan animasi yang sedang berjalan
	pcall(function()
		for _, t in ipairs(hum:GetPlayingAnimationTracks()) do t:AdjustSpeed(speed) end
	end)
end

local function startFly(char, hum)
	local root = char:FindFirstChild("HumanoidRootPart")
	if not root then return end
	flyClean()
	for _, s in ipairs(FLY_STATES) do
		if fly.saved[s] == nil then fly.saved[s] = hum:GetStateEnabled(s) end -- ingat state asli game
		hum:SetStateEnabled(s, false)
	end
	local animate = char:FindFirstChild("Animate")
	if animate then animate.Disabled = true end
	flyTracks(hum, 0)
	fly.bg = Instance.new("BodyGyro")
	fly.bg.Name, fly.bg.P = "VuDongFlyGyro", 9e4
	fly.bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
	fly.bg.CFrame = root.CFrame
	fly.bg.Parent = root
	fly.bv = Instance.new("BodyVelocity")
	fly.bv.Name = "VuDongFlyVel"
	fly.bv.Velocity = Vector3.zero
	fly.bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
	fly.bv.Parent = root
	hum.PlatformStand = true
	hum:ChangeState(Enum.HumanoidStateType.Swimming)
end

local function stopFly(hum)
	flyClean()
	fly.up, fly.down = false, false
	if hum and hum.Parent then
		for s, was in pairs(fly.saved) do hum:SetStateEnabled(s, was) end
		hum.PlatformStand = false
		local animate = hum.Parent:FindFirstChild("Animate")
		if animate then animate.Disabled = false end
		flyTracks(hum, 1)
		hum:ChangeState(Enum.HumanoidStateType.RunningNoPhysics)
	end
	table.clear(fly.saved)
end

-- satu-satunya loop Heartbeat: mengikuti MoveDirection, menjaga arah kamera, UP / DOWN
local function flyLoop(run)
	if fly.conn then fly.conn:Disconnect() fly.conn = nil end
	if not run then return end
	fly.conn = RunService.Heartbeat:Connect(function(dt)
		local char = player.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local root = char and char:FindFirstChild("HumanoidRootPart")
		if not (fly.on and hum and root and hum.Health > 0) then return end
		local k = math.min(dt * 60, 4) -- dinormalisasi ke 60 FPS (versi asli menggeser tetap tiap frame)
		local move = hum.MoveDirection
		if move.Magnitude > 0 then
			char:TranslateBy(move * fly.speed * k)
		end
		local vert = (fly.up and 1 or 0) - (fly.down and 1 or 0)
		if vert ~= 0 then
			root.CFrame = root.CFrame + Vector3.new(0, vert * k, 0)
		end
		local cam = workspace.CurrentCamera
		if fly.bg and cam then fly.bg.CFrame = cam.CFrame end
	end)
end

FlyAPI = {
	MIN = FLY_MIN, MAX = FLY_MAX, DEFAULT = FLY_DEFAULT,
	setSpeed = function(v) fly.speed = math.clamp(math.floor(v + 0.5), FLY_MIN, FLY_MAX) end, -- dipanggil oleh slider
	hold = function(dir, on) -- tombol melayang ▲ (dir = 1) / ▼ (dir = -1): tahan untuk naik / turun
		if dir > 0 then fly.up = on else fly.down = on end
	end,
	onChange = nil, -- bagian antarmuka menempel di sini untuk menampilkan / menyembunyikan tombol melayang
}

-- mati / respawn: otomatis diterapkan lagi ke karakter baru
player.CharacterAdded:Connect(function(char)
	local hum = char:WaitForChild("Humanoid", 10)
	if not hum then return end
	if godOn then startGod(hum) end
	if speedOn then startSpeed(hum) end
	if jumpOn then startJump(hum) end
	if fullOn then startFull(hum) end
	if invisOn then startInvis(char, hum) end
	if fly.on then -- jika Terbang aktif, terbang lagi dengan karakter baru
		table.clear(fly.saved)
		flyClean()
		char:WaitForChild("HumanoidRootPart", 5)
		char:WaitForChild("Animate", 3)
		if fly.on and player.Character == char then startFly(char, hum) end
	end
end)

-- LOGIKA: Boost FPS 120 (menurunkan grafis berat + membuka batas 120 FPS)
----------------------------------------------------------------
local Lighting = game:GetService("Lighting")
local FPS_TARGET = 120
local fpsOn = false
local fpsSaved = { effects = setmetatable({}, { __mode = "k" }), particles = setmetatable({}, { __mode = "k" }) }
local fpsConn, fpsLightingOrig, fpsQualityOrig = nil, nil, nil
local FPS_EFFECT_CLASSES = { "BlurEffect", "BloomEffect", "SunRaysEffect", "ColorCorrectionEffect", "DepthOfFieldEffect", "Atmosphere" }
local FPS_PARTICLE_CLASSES = { "ParticleEmitter", "Trail", "Smoke", "Fire", "Sparkles" }

local function fpsIsClass(obj, list)
	for _, c in ipairs(list) do
		if obj:IsA(c) then return true end
	end
	return false
end

local function fpsStrip(obj)
	if fpsIsClass(obj, FPS_EFFECT_CLASSES) then
		if fpsSaved.effects[obj] == nil then fpsSaved.effects[obj] = obj.Enabled end
		pcall(function() obj.Enabled = false end)
	elseif fpsIsClass(obj, FPS_PARTICLE_CLASSES) then
		if fpsSaved.particles[obj] == nil then fpsSaved.particles[obj] = obj.Enabled end
		pcall(function() obj.Enabled = false end)
	end
end

local function fpsCap(n)
	pcall(function()
		if setfpscap then setfpscap(n) end
	end)
end

local function startFps()
	-- 1) Buka batas FPS (hanya berjalan jika executor mendukung setfpscap)
	fpsCap(FPS_TARGET)
	-- 2) Turunkan grafis: matikan bayangan, efek post-processing, partikel/asap/api
	fpsLightingOrig = fpsLightingOrig or { shadows = Lighting.GlobalShadows }
	pcall(function() Lighting.GlobalShadows = false end)
	pcall(function()
		fpsQualityOrig = fpsQualityOrig or settings().Rendering.QualityLevel
		settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
	end)
	for _, d in ipairs(Lighting:GetDescendants()) do fpsStrip(d) end
	for _, d in ipairs(workspace:GetDescendants()) do fpsStrip(d) end
	-- 3) Efek baru yang muncul saat fitur aktif juga langsung dimatikan
	if fpsConn then fpsConn:Disconnect() end
	fpsConn = workspace.DescendantAdded:Connect(function(d)
		if fpsOn then fpsStrip(d) end
	end)
end

local function stopFps()
	if fpsConn then fpsConn:Disconnect() fpsConn = nil end
	fpsCap(60)
	if fpsLightingOrig then
		pcall(function() Lighting.GlobalShadows = fpsLightingOrig.shadows end)
		fpsLightingOrig = nil
	end
	if fpsQualityOrig then
		pcall(function() settings().Rendering.QualityLevel = fpsQualityOrig end)
		fpsQualityOrig = nil
	end
	for obj, was in pairs(fpsSaved.effects) do
		if obj.Parent then pcall(function() obj.Enabled = was end) end
	end
	for obj, was in pairs(fpsSaved.particles) do
		if obj.Parent then pcall(function() obj.Enabled = was end) end
	end
	table.clear(fpsSaved.effects)
	table.clear(fpsSaved.particles)
end

-- Fungsi bantu

local function new(class, props, parent)
	local o = Instance.new(class)
	for k, v in pairs(props) do
		o[k] = v
	end
	o.Parent = parent
	return o
end

local function corner(o, r)
	return new("UICorner", { CornerRadius = UDim.new(0, r) }, o)
end

local function stroke(o, color, th, trans)
	return new("UIStroke", { Color = color, Thickness = th or 1, Transparency = trans or 0 }, o)
end

local function pad(o, t, r, b, l)
	return new("UIPadding", {
		PaddingTop = UDim.new(0, t), PaddingRight = UDim.new(0, r),
		PaddingBottom = UDim.new(0, b), PaddingLeft = UDim.new(0, l),
	}, o)
end

local function list(o, gap, order)
	return new("UIListLayout", { Padding = UDim.new(0, gap), SortOrder = order or Enum.SortOrder.LayoutOrder }, o)
end

local function label(parent, props, keepSize)
	local p = {
		BackgroundTransparency = 1, Font = F_MED, TextColor3 = TEXT, TextSize = 16,
		TextXAlignment = Enum.TextXAlignment.Left,
	}
	for k, v in pairs(props) do
		p[k] = v
	end
	if not keepSize then
		p.TextSize = ts(p.TextSize) -- kalikan ukuran teks (keepSize = true berarti dibiarkan)
	end
	return new("TextLabel", p, parent)
end

local function metalStroke(o, thickness)
	local st = new("UIStroke", {
		Color = Color3.new(1, 1, 1), Thickness = thickness or 3, Transparency = 0.35,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	}, o)

	TweenService:Create(g, TweenInfo.new(10, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1), { Rotation = 360 }):Play()

end

local function toggle(parent, order, icon, text, default, group)
	local r = row(parent, order, icon, text, 240)
	State[text] = default
	if group then
		Groups[group] = Groups[group] or {}
		table.insert(Groups[group], text)
	end
	local sw = new("TextButton", {
		Text = "", AutoButtonColor = false, AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, -32, 0.5, 0), Size = UDim2.fromOffset(180, 96), BackgroundColor3 = OFF,

local function slider(parent, order, icon, text, minV, maxV, default, onChange)
	local SH = 260 -- tinggi baris: separuh atas = nama + angka, separuh bawah = slider
	local r = new("Frame", { BackgroundColor3 = ROW, BackgroundTransparency = 0.1, LayoutOrder = order, Size = UDim2.new(1, 0, 0, SH) }, parent)
	corner(r, 26)
	stroke(r, ROWLINE, 2, 0.6)
	label(r, { Text = icon, TextSize = 34, Font = F_BOLD, TextXAlignment = Enum.TextXAlignment.Center,
		Position = UDim2.fromOffset(14, 0), Size = UDim2.fromOffset(120, 130) })
	label(r, { Text = text, TextSize = 28, TextTruncate = Enum.TextTruncate.AtEnd,
		Position = UDim2.fromOffset(150, 0), Size = UDim2.new(1, -(150 + 360), 0, 130) })
	local valueLabel = label(r, { Text = tostring(default), TextSize = 30, Font = F_BOLD,
		TextXAlignment = Enum.TextXAlignment.Right, AnchorPoint = Vector2.new(1, 0),
		Position = UDim2.new(1, -32, 0, 0), Size = UDim2.fromOffset(320, 130) })
	themed(function() valueLabel.TextColor3 = theme.accent2 end)

	local BTN, CY = 110, 195 -- ukuran tombol -/+ dan titik tengah vertikal slider
	local function stepBtn(txt, anchorX, x)
		local b = new("TextButton", {
			Text = txt, TextSize = ts(30), Font = F_BOLD, TextColor3 = TEXT, AutoButtonColor = false,
			BackgroundColor3 = OFF, BorderSizePixel = 0, AnchorPoint = Vector2.new(anchorX, 0.5),
			Position = UDim2.new(anchorX, x, 0, CY), Size = UDim2.fromOffset(BTN, BTN),
		}, r)
		corner(b, 28)
		return b
	end
	local minus = stepBtn("−", 0, 150)
	local plus = stepBtn("+", 1, -32)

	-- area sentuh (lebih tinggi dari bar agar mudah digeser dengan jari)
	local hit = new("TextButton", {
		Text = "", AutoButtonColor = false, BackgroundTransparency = 1, BorderSizePixel = 0, AnchorPoint = Vector2.new(0, 0.5),
		Position = UDim2.new(0, 150 + BTN + 50, 0, CY), Size = UDim2.new(1, -(150 + BTN + 50 + 32 + BTN + 50), 0, 130),
	}, r)
	local track = new("Frame", {
		BackgroundColor3 = OFF, BorderSizePixel = 0, AnchorPoint = Vector2.new(0, 0.5),
		Position = UDim2.fromScale(0, 0.5), Size = UDim2.new(1, 0, 0, 28),
	}, hit)
	corner(track, 14)
	local fill = new("Frame", { BorderSizePixel = 0, Size = UDim2.fromScale(0, 1) }, track)
	corner(fill, 14)
	local knob = new("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(72, 72),
		BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0, ZIndex = 2,
	}, track)
	corner(knob, 36)
	themed(function() fill.BackgroundColor3 = theme.accent end)

	local value = math.clamp(default, minV, maxV)
	local function render()
		local a = (value - minV) / (maxV - minV)
		fill.Size = UDim2.fromScale(a, 1)
		knob.Position = UDim2.fromScale(a, 0.5)
		valueLabel.Text = tostring(value)
	end
	local function set(v)
		value = math.clamp(math.floor(v + 0.5), minV, maxV)
		render()
		onChange(value)
	end
	render()

	minus.Activated:Connect(function() set(value - 1) end)
	plus.Activated:Connect(function() set(value + 1) end)

	local function setFromX(x)
		local a = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
		set(minV + a * (maxV - minV))
	end
	local sliding = false
	local scroller = parent:FindFirstAncestorOfClass("ScrollingFrame")
	hit.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			sliding = true
			if scroller then scroller.ScrollingEnabled = false end -- saat menyeret, halaman tidak ikut scroll
			setFromX(input.Position.X)
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					sliding = false
					if scroller then scroller.ScrollingEnabled = true end
				end
			end)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			setFromX(input.Position.X)
		end
	end)
end

local function newPage(name)
	local p = new("ScrollingFrame", {
		Name = name, Visible = false, BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 16,
		CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y,
		ScrollingDirection = Enum.ScrollingDirection.Y,
		Position = UDim2.fromOffset(SIDE_W + MARG * 2, BODY_TOP), Size = UDim2.new(1, -(SIDE_W + MARG * 3), 1, -(BODY_TOP + BOT)),
	}, main)
	pad(p, 4, 30, 4, 4)
	list(p, 28)
	pages[name] = p
	return p
end

local function showTab(name)
	local page = pages[name]
	local wasVisible = page.Visible
	for n, p in pairs(pages) do
		if n ~= name and p.Visible then
			p:SetAttribute("SavedScrollY", p.CanvasPosition.Y) -- ingat posisi scroll tab yang akan ditinggalkan
		end
		p.Visible = (n == name)
	end
	if not wasVisible then
		-- kembali ke tab lama: kembalikan posisi scroll (pertama kali masuk tab ada di bagian atas)
		local y = page:GetAttribute("SavedScrollY") or 0
		page.CanvasPosition = Vector2.new(0, y)
		-- halaman yang baru tampil mungkin belum selesai menghitung panjang isi sehingga terjepit ke 0: set ulang beberapa frame pertama
		task.spawn(function()
			for _ = 1, 3 do
				task.wait()
				if not page.Visible or math.abs(page.CanvasPosition.Y - y) < 1 then return end
				page.CanvasPosition = Vector2.new(0, y)
			end
		end)
	end
	for n, b in pairs(navByName) do
		b.BackgroundTransparency = (n == name) and 0 or 1
	end
end

	local function textMatch(s, kws)
		if type(s) ~= "string" or s == "" then return false end
		local l = string.lower(s)
		for _, kw in ipairs(kws) do

		local c = A.cache[item]
		if c == true then return true end

		local function hit(x) return textMatch(x, A.keywords) end

				if hit(par.Name) then return true end
				par = par.Parent
			end
			for _, v in pairs(item:GetAttributes()) do
				if hit(v) then return true end
			end
			local n = 0
			for _, d in ipairs(item:GetDescendants()) do
				n += 1
				if n > 150 then break end
				if hit(d.Name) then return true end
				if d:IsA("ProximityPrompt") then
					if hit(d.ObjectText) or hit(d.ActionText) then return true end
				elseif d:IsA("TextLabel") or d:IsA("TextButton") then
					if hit(d.Text) then return true end
				elseif d:IsA("StringValue") then
					if hit(d.Value) then return true end
				end
			end
			return false
		end)
		if ok and res then
			A.cache[item] = true
			return true
		end
		A.cache[item] = os.clock()
		return false

	-- posisi tombol ambil (ProximityPrompt)
	local function promptPos(prompt, part)
		local par = prompt.Parent
		if par and par:IsA("Attachment") then return par.WorldPosition end
		if par and par:IsA("BasePart") then return par.Position end
		if par and par:IsA("Model") then return par:GetPivot().Position end
		return part.Position

	-- mengambil satu objek: prioritas ProximityPrompt (TAHAN) → ClickDetector → sentuh
	-- mengembalikan: (apakah berhasil diproses, apakah karakter dipindahkan)
	local function collect(root, part, item)
		local prompt = item:FindFirstChildWhichIsA("ProximityPrompt", true)
		if prompt and prompt.Enabled then
			local moved, hold = false, prompt.HoldDuration
			pcall(function() prompt.RequiresLineOfSight = false end) -- tetap bisa diambil meski terhalang objek lain
			local pos = promptPos(prompt, part)
			if (root.Position - pos).Magnitude > math.max(prompt.MaxActivationDistance - 3, 2) then
				-- berdiri terlalu jauh dari jangkauan: terbang mendekati objek (selesai pemindaian akan kembali ke pusat area)
				root.AssemblyLinearVelocity = Vector3.zero
				root.CFrame = CFrame.new(pos + Vector3.new(0, 2.5, 0))
				moved = true
				task.wait(0.15) -- tunggu tombol ambil muncul
			end
			if hold <= 0 and fpp then
				pcall(fpp, prompt) -- tombol sekali tekan, tidak perlu ditahan
			else
				-- tahan sungguhan: tahan sesuai HoldDuration, kunci karakter agar tidak jatuh keluar dari jangkauan
				pcall(function() root.Anchored = true end)
				pcall(function()
					prompt:InputHoldBegin()
					task.wait(hold + 0.1)
					prompt:InputHoldEnd()
				end)
				pcall(function() root.Anchored = false end)
				if fpp and prompt.Parent and prompt.Enabled then
					pcall(fpp, prompt) -- cadangan jika tahan belum berhasil
				end
			end
			return true, moved
		end
		local cd = item:FindFirstChildWhichIsA("ClickDetector", true)
		if cd and fcd then
			pcall(fcd, cd)
			return true, false
		end
		local ti = part:FindFirstChildWhichIsA("TouchInterest") or item:FindFirstChildWhichIsA("TouchInterest", true)
		if ti or (not part.CanCollide and part.Size.Magnitude <= 12) then
			if fti then
				pcall(fti, root, part, 0)
				pcall(fti, root, part, 1)
				return true, false
			end
			local back = root.CFrame -- tanpa firetouchinterest: sentuh cepat lalu kembali ke tempat semula
			root.CFrame = part.CFrame
			task.wait(0.1)
			root.CFrame = back
			return true, false
		end
		return false, false
	end

		local char = player.Character

		if root and (root.Position - A.center).Magnitude > A.radius then

				char:PivotTo(CFrame.new(A.center))
			end)
		end
	end

	-- satu kali pemindaian area A: semua objek di area → objek yang jenisnya cocok diambil satu per satu
	local function step(A)
		local char = player.Character
		local root = char and char:FindFirstChild("HumanoidRootPart")
		if not root then return end
		local ok, parts = pcall(workspace.GetPartBoundsInRadius, workspace, A.center, A.radius)
		if not ok then return end
		local seen, movedAny = {}, false
		for _, part in ipairs(parts) do
			if not A.on then break end
			if part.Parent and not part:IsDescendantOf(char) then
				local item = itemOf(part)
				if not seen[item] then
					seen[item] = true
					if (A.lastHit[item] or 0) + AUTO_COOLDOWN <= os.clock() and isTarget(A, item) then
						A.lastHit[item] = os.clock()
						local okc, handled, moved = pcall(collect, root, part, item)
						if okc and moved then movedAny = true end
					end
				end
			end
		end
		if movedAny and AUTO_GOTO then
			pcall(function() -- setelah selesai mengambil, kembali ke pusat area
				root.AssemblyLinearVelocity = Vector3.zero
				root.CFrame = CFrame.new(A.center)
			end)
		end
	end

	player.CharacterAdded:Connect(function()
		task.wait(1.5)
		if not AUTO_GOTO then return end
		for _, A in ipairs(autos) do
			if A.on then
				gotoCenter(A)
				break
			end
		end
	end)

	-- slider radius BERSAMA untuk semua area (geser = mengubah radius semua area)
	slider(autoCard, 1, "📏", "Radius Area Ambil", 5, 300, 50, function(v)
		for _, A in ipairs(autos) do A.radius = v end
	end)

	-- dibagi 3 kolom: tiap baris ada 3 tombol hidup / mati (ubah PER_ROW jika ingin jumlah tombol lain)

			center = cfg.center, radius = cfg.radius, keywords = cfg.keywords, on = false, run = 0, -- run bertambah tiap kali dihidupkan / dimatikan agar loop lama berhenti sendiri
			cache = setmetatable({}, { __mode = "k" }), lastHit = setmetatable({}, { __mode = "k" }),
		}
		table.insert(autos, A)
		State[cfg.name] = false

		if (i - 1) % PER_ROW == 0 then

				BackgroundTransparency = 1, LayoutOrder = 1 + math.ceil(i / PER_ROW),
				Size = UDim2.new(1, 0, 0, ROW_H),
			}, autoCard)
			new("UIListLayout", {
				FillDirection = Enum.FillDirection.Horizontal, Padding = UDim.new(0, GAP),
				VerticalAlignment = Enum.VerticalAlignment.Center, SortOrder = Enum.SortOrder.LayoutOrder,
			}, rowFrame)
		end
		local b = new("TextButton", {
			Text = "", TextSize = ts(24), Font = F_BOLD, TextColor3 = Color3.new(1, 1, 1), AutoButtonColor = true,
			TextTruncate = Enum.TextTruncate.AtEnd, LayoutOrder = (i - 1) % PER_ROW,
			BackgroundColor3 = OFF,
			Size = UDim2.new(1 / PER_ROW, -(GAP * (PER_ROW - 1)) / PER_ROW, 0, ROW_H - 20),
		}, rowFrame)
		corner(b, 24)
		themed(function() -- hidup = warna tema + tanda ✅, mati = warna gelap + 💎
			local on = State[cfg.name]
			b.BackgroundColor3 = on and theme.accent or OFF
			b.Text = (on and "✅ " or "💎 ") .. cfg.short
		end)
		b.Activated:Connect(function()
			State[cfg.name] = not State[cfg.name]
			local on = State[cfg.name]
			b.BackgroundColor3 = on and theme.accent or OFF
			b.Text = (on and "✅ " or "💎 ") .. cfg.short
			setStatus((on and "Aktif: " or "Nonaktif: ") .. cfg.name)
			local cb = Callbacks[cfg.name]
			if cb then task.spawn(cb, on) end
		end)

		Callbacks[cfg.name] = function(on)
			A.on = on
			A.run += 1
			local run = A.run
			if not on then return end
			table.clear(A.cache)
			table.clear(A.lastHit)
			if AUTO_GOTO and not lock then gotoCenter(A) end
			while A.on and run == A.run do
				while lock and A.on and run == A.run do task.wait(0.05) end -- tunggu area lain selesai mengambil
				if A.on and run == A.run then
					lock = true
					pcall(step, A)
					lock = false
				end
				task.wait(AUTO_TICK)
			end
		end
	end
end

-- path yang bisa dipakai di script: workspace["Map"]["Part"] atau game:GetService("X")["Y"]
local function pathOf(inst)
	local parts, cur = {}, inst
	while cur and cur ~= game do
		table.insert(parts, 1, cur)
		cur = cur.Parent
	end
	if cur ~= game then return inst:GetFullName() .. " (bukan bagian dari game)" end
	if #parts == 0 then return "game" end
	local s
	if parts[1] == workspace then
		s = "workspace"
	else
		s = string.format("game:GetService(%q)", parts[1].ClassName)
	end
	for i = 2, #parts do
		s = s .. string.format("[%q]", parts[i].Name)
	end
	return s
end

local function getExecRows()
	if execRows then return execRows end
	local ok, env = pcall(function() return getfenv() end)
	if not ok or type(env) ~= "table" then env = {} end
	local name = "tidak diketahui (Studio atau bukan executor)"
	if env.identifyexecutor then
		local ok2, n = pcall(env.identifyexecutor)
		if ok2 and n then name = tostring(n) end
	end
	local has, miss = {}, {}
	for _, f in ipairs(EXEC_FUNCS) do
		if env[f] ~= nil then table.insert(has, f) else table.insert(miss, f) end
	end
	execRows = {
		{ "Executor", name },
		{ "Fungsi tersedia", #has > 0 and table.concat(has, ", ") or "(tidak ada)" },
		{ "Fungsi tidak ada", #miss > 0 and table.concat(miss, ", ") or "(tidak ada yang kurang)" },
	}
	return execRows
end

local function collectDiag()
	local rows = {}
	local function add(k, v) table.insert(rows, { k, tostring(v) }) end
	add("FPS", diagOn and diagFps or "nonaktif")
	local ping = "?"
	pcall(function() ping = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue() + 0.5) .. " ms" end)
	add("Ping", ping)
	local mem = "?"
	pcall(function() mem = string.format("%.0f MB", Stats:GetTotalMemoryUsageMb()) end)
	add("Memori", mem)
	add("Pemain", #Players:GetPlayers() .. "/" .. Players.MaxPlayers)
	add("PlaceId", game.PlaceId)
	local char = player.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if root then
		add("Posisi", string.format("%.1f, %.1f, %.1f", root.Position.X, root.Position.Y, root.Position.Z))
	else
		add("Posisi", "belum ada karakter")
	end
	local hum = getHumanoid()
	if hum then
		add("Darah", string.format("%.0f / %.0f", hum.Health, hum.MaxHealth))
		add("WalkSpeed", hum.WalkSpeed)
		add("Lompat", hum.UseJumpPower and ("JumpPower " .. hum.JumpPower) or ("JumpHeight " .. hum.JumpHeight))
		add("Status", hum:GetState().Name)
		add("RigType", hum.RigType.Name)
	end
	if lastTarget and lastTarget.Parent then
		add("Objek terakhir diklik", pathOf(lastTarget) .. "  [" .. lastTarget.ClassName .. "]")
	else
		add("Objek terakhir diklik", "belum klik objek apa pun")
	end
	for _, r in ipairs(getExecRows()) do table.insert(rows, r) end
	return rows
end

	for _, r in ipairs(rows) do

	end
	return table.concat(out, "\n")
end

local logDirty = false
local function renderLog()
	logDirty = false
	if #logLines == 0 then
		logLabel.Text = LOG_IDLE
		return
	end
	local out = {}
	for i = math.max(1, #logLines - LOG_SHOW + 1), #logLines do
		local l = logLines[i]
		table.insert(out, string.format('<font color="%s">%s</font>', l.color, esc(l.text)))
	end
	logLabel.Text = table.concat(out, "\n")
end

local function addLog(text, mtype)
	text = tostring(text)
	local cut = utf8.offset(text, 221)
	if cut then text = text:sub(1, cut - 1) .. "…" end
	table.insert(logLines, { text = os.date("%H:%M:%S") .. "  " .. text, color = LOG_COLORS[mtype] or "#cdd9f5" })
	if #logLines > LOG_MAX then table.remove(logLines, 1) end
	if not logDirty then
		logDirty = true
		task.defer(renderLog)
	end
end

Callbacks["Catat Log Konsol"] = function(on)
	logOn = on
	if logConn then
		logConn:Disconnect()
		logConn = nil
	end
	if not on then return end
	table.clear(logLines)
	pcall(function()
		local hist = LogService:GetLogHistory()
		for i = math.max(1, #hist - 10), #hist do
			addLog(hist[i].message, hist[i].messageType)
		end
	end)
	logConn = LogService.MessageOut:Connect(addLog)
	renderLog()
end

-- PINDAI MAP (halaman Pengaturan): membaca info + posisi objek di SELURUH map (workspace)
-- Dipakai agar menulis script lebih mudah: tahu nama objek, jenisnya, lokasinya, path-nya, bisa diklik / diambil atau tidak...
-- Dibungkus dalam blok do ... end agar tidak memakai variabel local tambahan di tingkat luar (Luau membatasi 200).
-- Atur SCAN_MAX / SCAN_PAGE / SCAN_CHEST tepat di bawah jika perlu.
-- CATATAN: jika game mengaktifkan StreamingEnabled, objek yang jauh (belum dimuat) belum bisa dipindai - dekati lalu pindai ulang.
----------------------------------------------------------------
do
	local SCAN_MAX = 30000 -- jumlah objek maksimal yang dicatat tiap pemindaian
	local SCAN_PAGE = 10 -- jumlah objek yang tampil di kotak tiap halaman
	local SCAN_COPY_MAX = 3000 -- jumlah baris maksimal saat klik "Salin Hasil" (Simpan File menyimpan semuanya)
	local SCAN_MARKS = 80 -- jumlah label 3D maksimal (prioritas objek terdekat denganmu)
	local SCAN_CHEST = { "chest", "peti", "rương", "ruong" } -- huruf kecil; nama yang mengandung kata ini = jenis "Peti"
	local SCAN_CHEST_IGNORE = { "plate", "armor", "armour" }
	local CATS = { "Semua", "Model", "Part", "Tool", "Interaktif", "NPC", "Peti", "Spawn/Kursi" }
	local SCAN_IDLE = 'Belum dipindai. Klik "Pindai Seluruh Map" untuk membaca semua objek di workspace: nama, jenis, path, posisi, ukuran... (Radius 0 = seluruh map).'

	local scanCard = card(setCard.Parent, 3, "🗺️", "Pindai Map", "Baca & pindai posisi objek di seluruh map", Color3.fromRGB(52, 211, 153))
	metalCard(scanCard)

	local all, view, marks = {}, {}, {} -- all = semua objek yang dipindai, view = objek setelah difilter, marks = label 3D
	local summaryLines = {}
	local catFilter, radius, curPage, busy = "Semua", 0, 1, false
	local scanLabel -- kotak teks hasil (dibuat di bawah)

	local function fmtV(v)
		return string.format("%.1f, %.1f, %.1f", v.X, v.Y, v.Z)
	end

	local function getRoot()
		local ch = player.Character
		return ch and ch:FindFirstChild("HumanoidRootPart")
	end

	local function isChest(name)
		local l = string.lower(name)
		for _, bad in ipairs(SCAN_CHEST_IGNORE) do
			if string.find(l, bad, 1, true) then return false end
		end
		for _, kw in ipairs(SCAN_CHEST) do
			if string.find(l, kw, 1, true) then return true end
		end
		return false
	end

	-- klasifikasi satu objek (nil = dilewati)
	local function classify(inst)
		if inst:IsA("ProximityPrompt") or inst:IsA("ClickDetector") then return "Interaktif" end
		if inst:IsA("Tool") then return "Tool" end
		if inst:IsA("Model") then
			if inst:FindFirstChildOfClass("Humanoid") then return "NPC" end
			if isChest(inst.Name) then return "Peti" end
			return "Model"
		end
		if inst:IsA("BasePart") then
			if inst:IsA("Terrain") then return nil end
			if inst:IsA("SpawnLocation") or inst:IsA("Seat") or inst:IsA("VehicleSeat") then return "Spawn/Kursi" end
			if isChest(inst.Name) then return "Peti" end
			return "Part"
		end
		return nil
	end

	-- posisi objek (Prompt / Click mengikuti objek induk)
	local function posOf(inst)
		local ok, p = pcall(function()
			if inst:IsA("BasePart") then return inst.Position end
			if inst:IsA("Model") then return inst:GetPivot().Position end
			if inst:IsA("Tool") then
				local h = inst:FindFirstChild("Handle")
				if h and h:IsA("BasePart") then return h.Position end
			end
			local par = inst.Parent
			while par and par ~= workspace do
				if par:IsA("BasePart") then return par.Position end
				if par:IsA("Model") then return par:GetPivot().Position end
				par = par.Parent
			end
			return nil
		end)
		if ok then return p end
		return nil
	end

	-- info tambahan (dihitung saat perlu ditampilkan, tidak saat memindai agar ringan)
	local function extraOf(inst)
		local t = {}
		pcall(function()
			if inst:IsA("BasePart") then
				table.insert(t, "Size " .. fmtV(inst.Size))
				table.insert(t, inst.Anchored and "Anchored" or "Free")
				table.insert(t, inst.CanCollide and "CanCollide" or "NoCollide")
				if inst.Transparency >= 1 then table.insert(t, "tak terlihat") end
			elseif inst:IsA("Model") then
				local hum = inst:FindFirstChildOfClass("Humanoid")
				if hum then
					table.insert(t, string.format("Darah %.0f/%.0f", hum.Health, hum.MaxHealth))
					if Players:GetPlayerFromCharacter(inst) then table.insert(t, "pemain") end
				end
				if inst.PrimaryPart then table.insert(t, "PrimaryPart=" .. inst.PrimaryPart.Name) end
				table.insert(t, #inst:GetChildren() .. " anak")
			elseif inst:IsA("ProximityPrompt") then
				table.insert(t, string.format("ActionText=%q ObjectText=%q Hold=%.1fs Jangkauan=%.0f %s", inst.ActionText, inst.ObjectText,
					inst.HoldDuration, inst.MaxActivationDistance, inst.Enabled and "Aktif" or "Mati"))
			elseif inst:IsA("ClickDetector") then
				table.insert(t, string.format("Jangkauan=%.0f", inst.MaxActivationDistance))
			elseif inst:IsA("Tool") then
				table.insert(t, inst:FindFirstChild("Handle") and "ada Handle" or "tanpa Handle")
			end
		end)
		pcall(function()
			local n = 0
			for k, v in pairs(inst:GetAttributes()) do
				n += 1
				if n > 4 then break end
				table.insert(t, k .. "=" .. tostring(v))
			end
		end)
		return table.concat(t, " | ")
	end

	-- satu baris lengkap (untuk salin / simpan file / konsol)
	local function fullLine(i, e)
		local parts = {
			string.format("#%d [%s] %s", i, e.cat, e.cls),
			pathOf(e.inst),
			e.pos and string.format("Vector3.new(%.2f, %.2f, %.2f)", e.pos.X, e.pos.Y, e.pos.Z) or "posisi tidak diketahui",
		}
		if e.dist then table.insert(parts, string.format("jarak %.1f stud", e.dist)) end
		local x = extraOf(e.inst)
		if x ~= "" then table.insert(parts, x) end
		return table.concat(parts, " | ")
	end

	local function buildReport(maxLines)
		local lines = { "=== Vu Dong Script - Pindai Map (PlaceId " .. tostring(game.PlaceId) .. ") ===" }
		for _, s in ipairs(summaryLines) do table.insert(lines, s) end
		table.insert(lines, string.format("Filter: %s | Radius: %s | %d hasil", catFilter,
			radius > 0 and (radius .. " stud") or "seluruh map", #view))
		table.insert(lines, "--------------------------------------------")
		for i = 1, math.min(#view, maxLines) do
			table.insert(lines, fullLine(i, view[i]))
		end
		if #view > maxLines then
			table.insert(lines, "... masih ada " .. (#view - maxLines) .. " baris lagi (pakai tombol Simpan File untuk mengambil semuanya)")
		end
		return table.concat(lines, "\n")
	end

	-- tampilkan hasil di kotak teks
	local function render()
		if not scanLabel then return end
		if #all == 0 then
			scanLabel.Text = esc(SCAN_IDLE)
			return
		end
		local out = {}
		for _, s in ipairs(summaryLines) do table.insert(out, esc(s)) end
		local totalPages = math.max(1, math.ceil(#view / SCAN_PAGE))
		curPage = math.clamp(curPage, 1, totalPages)
		table.insert(out, string.format("── Halaman %d/%d · %d hasil · filter: %s · radius: %s ──",
			curPage, totalPages, #view, esc(catFilter), radius > 0 and (radius .. " stud") or "seluruh map"))
		for i = (curPage - 1) * SCAN_PAGE + 1, math.min(#view, curPage * SCAN_PAGE) do
			local e = view[i]
			local where = e.pos and fmtV(e.pos) or "posisi tidak diketahui"
			local dist = e.dist and string.format(" · jarak %.0f", e.dist) or ""
			table.insert(out, string.format("#%d [%s] %s\n      %s%s",
				i, esc(e.cat), esc(e.name), esc(where), dist))
		end
		if #view == 0 then table.insert(out, "(tidak ada objek yang cocok dengan filter)") end
		scanLabel.Text = table.concat(out, "\n")
	end

	-- filter berdasarkan jenis + radius, urutkan dari dekat ke jauh
	local function applyFilter()
		table.clear(view)
		local root = getRoot()
		local rp = root and root.Position
		for _, e in ipairs(all) do
			if catFilter == "Semua" or e.cat == catFilter then
				local d = (rp and e.pos) and (e.pos - rp).Magnitude or nil
				if radius <= 0 or (d and d <= radius) then
					e.dist = d
					table.insert(view, e)
				end
			end
		end
		table.sort(view, function(a, b)
			local da, db = a.dist or math.huge, b.dist or math.huge
			if da ~= db then return da < db end
			return a.name < b.name
		end)
	end

	-- label 3D di dalam game: tampilkan nama + koordinat pada objek terdekat denganmu (menembus dinding)
	local function clearMarks()
		for _, g in ipairs(marks) do g:Destroy() end
		table.clear(marks)
	end

	local function adorneeOf(inst)
		if inst:IsA("BasePart") then return inst end
		if inst:IsA("Model") then return inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart", true) end
		if inst:IsA("Tool") then
			local h = inst:FindFirstChild("Handle")
			if h and h:IsA("BasePart") then return h end
		end
		local par = inst.Parent
		while par and par ~= workspace do
			if par:IsA("BasePart") then return par end
			if par:IsA("Model") then return par.PrimaryPart or par:FindFirstChildWhichIsA("BasePart", true) end
			par = par.Parent
		end
		return nil
	end

	local function drawMarks()
		clearMarks()
		if not State["Tampilkan Label Posisi"] then return end
		local gui = player:FindFirstChildOfClass("PlayerGui")
		if not gui then return end
		for i = 1, math.min(#view, SCAN_MARKS) do
			local e = view[i]
			local ok, part = pcall(adorneeOf, e.inst)
			if ok and part and part.Parent then
				local bb = Instance.new("BillboardGui")
				bb.Name = "VuDongMapMark"
				bb.Adornee = part
				bb.AlwaysOnTop = true
				bb.ResetOnSpawn = false
				bb.Size = UDim2.fromOffset(220, 46)
				bb.StudsOffset = Vector3.new(0, 2, 0)
				bb.MaxDistance = 2000
				local t = Instance.new("TextLabel")
				t.BackgroundTransparency = 1
				t.Size = UDim2.fromScale(1, 1)
				t.Font = Enum.Font.SourceSansBold
				t.TextSize = 14
				t.TextWrapped = true
				t.TextColor3 = Color3.fromRGB(110, 231, 183)
				t.TextStrokeTransparency = 0
				t.Text = string.format("#%d %s\n%s", i, e.name, e.pos and fmtV(e.pos) or "")
				t.Parent = bb
				bb.Parent = gui
				table.insert(marks, bb)
			end
		end
	end

	-- ubah filter / radius: dikumpulkan, tunggu 0.25 detik baru dihitung (geser slider tidak lag)
	local refreshToken = 0
	local function refreshSoon()
		refreshToken += 1
		local my = refreshToken
		task.delay(0.25, function()
			if my == refreshToken and #all > 0 then
				applyFilter()
				render()
				drawMarks()
			end
		end)
	end

	-- PINDAI SELURUH MAP
	local function runScan()
		if busy then
			setStatus("Sedang memindai, tunggu sebentar...")
			return
		end
		busy = true
		setStatus("Sedang memindai map...")
		local t0 = os.clock()
		table.clear(all)

		-- lewati bagian tubuh di dalam karakter pemain (agar tidak berantakan), karakternya sendiri tetap dicatat
		local skip = {}
		for _, p in ipairs(Players:GetPlayers()) do
			local ch = p.Character
			if ch then
				for _, d in ipairs(ch:GetDescendants()) do skip[d] = true end
			end
		end

		local counts, capped = {}, false
		local minP, maxP, sumP, nPos = nil, nil, Vector3.zero, 0
		local ok, err = pcall(function()
			local n = 0
			for _, inst in ipairs(workspace:GetDescendants()) do
				n += 1
				if n % 1500 == 0 then task.wait() end -- beri jeda frame agar game tidak freeze
				if not skip[inst] then
					local cat = classify(inst)
					if cat then
						local pos = posOf(inst)
						table.insert(all, { inst = inst, name = inst.Name, cls = inst.ClassName, cat = cat, pos = pos })
						counts[cat] = (counts[cat] or 0) + 1
						if pos then
							nPos += 1
							sumP += pos
							minP = minP and Vector3.new(math.min(minP.X, pos.X), math.min(minP.Y, pos.Y), math.min(minP.Z, pos.Z)) or pos
							maxP = maxP and Vector3.new(math.max(maxP.X, pos.X), math.max(maxP.Y, pos.Y), math.max(maxP.Z, pos.Z)) or pos
						end
						if #all >= SCAN_MAX then
							capped = true
							break
						end
					end
				end
			end
		end)
		busy = false
		if not ok then
			setStatus("Error pindai map: " .. tostring(err))
			addLog("Error pindai map: " .. tostring(err), Enum.MessageType.MessageError)
			return
		end

		table.clear(summaryLines)
		table.insert(summaryLines, string.format("Berhasil memindai %d objek di workspace (%.1f detik)", #all, os.clock() - t0))
		local cs = {}
		for _, name in ipairs(CATS) do
			if counts[name] then table.insert(cs, name .. " " .. counts[name]) end
		end
		table.insert(summaryLines, #cs > 0 and table.concat(cs, " · ") or "(tidak ada objek terlihat)")
		if minP and maxP then
			table.insert(summaryLines, string.format("Rentang map: X %.0f→%.0f | Y %.0f→%.0f | Z %.0f→%.0f",
				minP.X, maxP.X, minP.Y, maxP.Y, minP.Z, maxP.Z))
			local c = sumP / nPos
			table.insert(summaryLines, string.format("Titik pusat rata-rata: %s", fmtV(c)))
		end
		local root = getRoot()
		if root then table.insert(summaryLines, "Posisimu saat memindai: " .. fmtV(root.Position)) end
		if workspace.StreamingEnabled then
			table.insert(summaryLines, "⚠ Game mengaktifkan StreamingEnabled: objek jauh yang belum dimuat belum bisa dipindai")
		end
		if capped then
			table.insert(summaryLines, "⚠ Mencapai batas " .. SCAN_MAX .. " objek - naikkan SCAN_MAX jika perlu memindai semuanya")
		end

		curPage = 1
		applyFilter()
		render()
		drawMarks()
		addLog(summaryLines[1], Enum.MessageType.MessageInfo)
		setStatus("Pemindaian selesai: " .. #all .. " objek")
	end

	-- antarmuka: satu fitur tiap baris
	actionRow(scanCard, 1, "🗺️", "Pindai Seluruh Map", "Pindai", runScan)

	dropdown(scanCard, 2, "🔎", "Filter Jenis Objek", CATS, "Semua", function(name)
		catFilter = name
		curPage = 1
		refreshSoon()
		setStatus("Filter: " .. name)
	end)

	slider(scanCard, 3, "📏", "Radius", 0, 2000, 0, function(v)
		radius = v
		curPage = 1
		refreshSoon()
	end)

	scanLabel = textPanel(scanCard, 4)
	scanLabel.Text = esc(SCAN_IDLE)

	actionRow(scanCard, 5, "◀️", "Halaman Sebelumnya", "◀", function()
		local totalPages = math.max(1, math.ceil(#view / SCAN_PAGE))
		curPage = curPage - 1
		if curPage < 1 then curPage = totalPages end
		render()
	end)

	actionRow(scanCard, 6, "▶️", "Halaman Berikutnya", "▶", function()
		local totalPages = math.max(1, math.ceil(#view / SCAN_PAGE))
		curPage = curPage + 1
		if curPage > totalPages then curPage = 1 end
		render()
	end)

	toggle(scanCard, 7, "🏷️", "Tampilkan Label Posisi", false, "Pengaturan")
	Callbacks["Tampilkan Label Posisi"] = function(on)
		if on then drawMarks() else clearMarks() end
	end

	actionRow(scanCard, 8, "📋", "Salin Hasil", "Salin", function()
		if #all == 0 then
			setStatus("Pindai map dulu")
			return
		end
		copyText(buildReport(SCAN_COPY_MAX), "Berhasil menyalin " .. math.min(#view, SCAN_COPY_MAX) .. " baris hasil pindai")
	end)

	actionRow(scanCard, 9, "💾", "Simpan File .txt", "Simpan", function()
		if #all == 0 then
			setStatus("Pindai map dulu")
			return
		end
		local fname = "VuDong_MapScan_" .. tostring(game.PlaceId) .. ".txt"
		if pcall(writefile, fname, buildReport(math.huge)) then
			setStatus("Berhasil menyimpan " .. #view .. " baris ke file " .. fname)
			addLog("Hasil pindai disimpan: " .. fname, Enum.MessageType.MessageInfo)
		else
			setStatus("Executor tidak mendukung writefile - pakai tombol Salin")
		end
	end)

	actionRow(scanCard, 10, "🎯", "Salin Objek Terdekat", "Salin", function()
		local e = view[1]
		if not e then
			setStatus("Belum ada hasil - pindai map dulu")
			return
		end
		copyText(fullLine(1, e), "Objek terdekat berhasil disalin: " .. e.name)
	end)

	actionRow(scanCard, 11, "🖨️", "Cetak ke Konsol", "Cetak", function()
		if #view == 0 then
			setStatus("Belum ada hasil - pindai map dulu")
			return
		end
		local n = math.min(#view, 300)
		print("=== Vu Dong Script - Pindai Map: " .. #view .. " hasil (mencetak " .. n .. " baris pertama) ===")
		for i = 1, n do print(fullLine(i, view[i])) end
		setStatus("Berhasil mencetak " .. n .. " baris ke konsol")
	end)

	actionRow(scanCard, 12, "🧹", "Hapus Hasil Pindai", "Hapus", function()

		table.clear(summaryLines)

		curPage = 1

		setStatus("Hasil pindai dihapus")
	end)

----------------------------------------------------------------
-- TEKS SELURUH MENU: animasi warna (DEFAULT MATI agar nyaman di mata)
-- - Berlaku untuk SEMUA TextLabel / TextButton / TextBox di menu, termasuk teks yang dibuat belakangan
--   (notifikasi, daftar pindai map...). Memakai palet BLINK_COLORS dan ritme BLINK_STEP yang sama.
-- - Hanya SATU loop untuk semua teks (tanpa Tween terpisah per teks) dan teks tersembunyi dilewati sehingga ringan.
-- - Tiap teks bergeser 1 warna sehingga terbentuk efek warna bergerak.
-- - Tidak diubah: teks di log Diagnostik (warna tetap menurut level error / peringatan) dan label 3D ESP / Pindai Map di dunia game.
-- - Ditaruh di fungsi tersendiri (task.spawn) agar TIDAK menambah variabel local di tingkat paling luar (batas 200 variabel).
----------------------------------------------------------------
task.spawn(function()
	local TEXT_ANIMATE = false -- true = teks berganti-ganti warna (BLINK_COLORS); false = teks diam & nyaman di mata
	local TEXT_FADE = 0.6    -- bagian tiap ritme yang dipakai untuk beralih ke warna berikut (0 = ganti warna instan)
	local TEXT_OUTLINE = 0.7 -- transparansi garis tepi teks hitam tipis agar mudah dibaca (1 = tanpa garis tepi)
	local TEXT_FPS = 20      -- jumlah pembaruan warna per detik

	local n = #BLINK_COLORS
	local tracked = setmetatable({}, { __mode = "k" }) -- [teks] = { fase warna, warna terakhir yang diset }
	local count = 0

	local function track(d)
		if tracked[d] then return end
		if d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox") then
			count += 1
			tracked[d] = { count % n, nil }
			if TEXT_OUTLINE < 1 and d.TextStrokeTransparency >= 1 then
				d.TextStrokeColor3 = Color3.new(0, 0, 0)
				d.TextStrokeTransparency = TEXT_OUTLINE
			end
		end
	end
	for _, d in ipairs(gui:GetDescendants()) do track(d) end
	gui.DescendantAdded:Connect(track)

	-- true = tampil, false = tersembunyi (dirinya atau bingkai induk), nil = sudah dihapus dari menu
	local function shown(p, memo)
		if p == gui then return true end
		if p == nil then return nil end
		local m = memo[p]
		if m ~= nil then return m end
		local r
		if p:IsA("GuiObject") and not p.Visible then
			r = false
		else
			r = shown(p.Parent, memo)
		end
		memo[p] = r
		return r
	end

	-- warna fase `phase` pada waktu t: bertahan di satu warna lalu beralih halus ke warna berikut
	local function colorAt(t, phase)
		local pos = t / BLINK_STEP + phase
		local i = math.floor(pos)
		local a = BLINK_COLORS[i % n + 1]
		if TEXT_FADE <= 0 then return a end
		local k = math.clamp((pos - i) / TEXT_FADE, 0, 1)
		k = k * k * (3 - 2 * k)
		return a:Lerp(BLINK_COLORS[(i + 1) % n + 1], k)
	end

	local cols, memo = {}, {}
	local acc, t0, conn = 0, os.clock(), nil
	conn = RunService.Heartbeat:Connect(function(dt)
		acc += dt
		if acc < 1 / TEXT_FPS then return end
		acc = 0
		if not gui.Parent then conn:Disconnect() return end
		if not TEXT_ANIMATE then return end
		local t = os.clock() - t0
		for phase = 0, n - 1 do cols[phase] = colorAt(t, phase) end
		table.clear(memo)
		for d, rec in pairs(tracked) do
			local s = shown(d, memo)
			if s == nil then
				tracked[d] = nil
			elseif s then
				local c = cols[rec[1]]
				if rec[2] ~= c then
					rec[2] = c
					d.TextColor3 = c
				end
			end
		end
	end)
end)
